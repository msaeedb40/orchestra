Release name:    {{ .Release.Name }}
Namespace:       {{ .Release.Namespace }}
Docker mode:     {{ .Values.docker.mode }}

1. Wait for the daemon to become ready:

   kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "orchestra.fullname" . }}

2. Forward the gRPC port locally:

   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "orchestra.fullname" . }} 50051:50051

3. Verify the daemon is reachable:

   orchestra system ping

4. Create your first cluster:

   orchestra cluster create dev \
     --control-plane 1 \
     --workers 2 \
     --cni orchestranet

5. Watch cluster status:

   orchestra cluster list
   orchestra cluster get dev
